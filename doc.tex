\subsection{قراردادهای نامگذاری}

\begin{itemize}[rightmargin=1cm]
  \item \textbf{بک‌اند (\lr{Django/Python}):}
    \begin{itemize}
      \item نام فایل‌ها و ماژول‌ها: \lr{snake\_case} (مثال: \lr{case\_views.py})
      \item نام کلاس‌ها: \lr{PascalCase} (مثال: \lr{ComplaintReview})
      \item نام متغیرها و توابع: \lr{snake\_case} (مثال: \lr{assigned\_detective})
      \item نام ثابت‌ها: \lr{UPPER\_SNAKE\_CASE} (مثال: \lr{Status.PENDING\_CADET})
      \item اپلیکیشن‌های \lr{Django}: نام کوتاه و جمع (مثال: \lr{cases}، \lr{users}، \lr{evidence})
    \end{itemize}
  \item \textbf{فرانت‌اند (\lr{React/TypeScript}):}
    \begin{itemize}
      \item نام کامپوننت‌ها: \lr{PascalCase} (مثال: \lr{CaseDetailPage})
      \item نام \lr{hook}ها: پیشوند \lr{use} + \lr{camelCase} (مثال: \lr{useAuthStore})
      \item نام فایل‌های کامپوننت: \lr{PascalCase.tsx}
      \item نام فایل‌های ابزاری: \lr{camelCase.ts}
      \item نام متغیرها و توابع: \lr{camelCase}
    \end{itemize}
\end{itemize}

\section{نحوه مدیریت پروژه}
% ─── این بخش را خودتان تکمیل کنید ───────────────────────────────────────────

در پروژه اکثرا تسک ها به صورت مشارکتی بین اعضای تیم بر اساس علاقه‌مندی ها و توانمندی ها و با همفکری هم در گروه تلگرامی تقسیم می‌شدند که این موضوع به پیشبرد هر چه بهتر پروژه کمک شایانی کرد.

\section{موجودیت‌های کلیدی سامانه}

در این بخش موجودیت‌های اصلی سامانه به همراه دلیل وجود آن‌ها توضیح داده می‌شود.

% ── 4.1 User ──────────────────────────────────────────────────────────────
\subsection{\lr{User} — کاربر}

\textbf{دلیل وجود:}
تمام افراد سامانه (شهروند، کادت، افسر، کارآگاه، سروان، کاپیتان، رئیس، پزشک قانونی، قاضی) از یک مدل کاربری واحد مشتق می‌شوند.
این موجودیت \lr{AbstractUser} جنگو را گسترش می‌دهد و فیلدهای اضافه‌ای مانند \lr{national\_id}، \lr{phone\_number} و \lr{email} را اجباری می‌کند.
مدیریت نقش‌ها از طریق سیستم \lr{Group} جنگو (\lr{RBAC} پویا) انجام می‌شود که افزودن و حذف نقش را بدون تغییر مدل ممکن می‌سازد.

% ── 4.2 RoleProfile ───────────────────────────────────────────────────────
\subsection{\lr{RoleProfile} — پروفایل نقش}

\textbf{دلیل وجود:}
هر \lr{Group} جنگو نماینده‌ی یک نقش است. \lr{RoleProfile} تنها یک فیلد توضیحات به آن اضافه می‌کند تا تعریف نقش‌ها در پنل مدیریت مستند باشد.

% ── 4.3 Case ──────────────────────────────────────────────────────────────
\subsection{\lr{Case} — پرونده}

\textbf{دلیل وجود:}
محور اصلی سامانه است. هر جریان کاری (تحقیق، دستگیری، صدور حکم) به یک پرونده متصل است.
این موجودیت یک ماشین وضعیت (\lr{State Machine}) با ۱۲ حالت مشخص دارد
(\lr{Draft} تا \lr{Closed}/\lr{Void}) که انتقال‌های غیرمجاز را پیشگیری می‌کند.
همچنین سطح اهمیت پرونده (\lr{Severity}) تعیین می‌کند آیا کیس نیاز به تأیید رئیس پلیس دارد یا خیر.

% ── 4.4 Complaint ─────────────────────────────────────────────────────────
\subsection{\lr{Complaint} — شکایت}

\textbf{دلیل وجود:}
اولین نقطه‌ی ورود اطلاعات از جانب شهروند است.
منطق سه ضربه (\lr{three-strike}): اگر یک شکایت سه بار توسط بازبین برگردانده شود، به‌صورت خودکار باطل (\lr{Void}) می‌شود.
این قانون مستقیماً در متد \lr{register\_invalid\_attempt} پیاده‌سازی شده است.

% ── 4.5 ComplaintReview ───────────────────────────────────────────────────
\subsection{\lr{ComplaintReview} — بررسی شکایت}

\textbf{دلیل وجود:}
فرایند تأیید شکایت دو مرحله دارد: بررسی کادت و بررسی افسر.
این موجودیت تاریخچه‌ی تمام تصمیم‌ها را نگه می‌دارد و رد پای مسئولیت (\lr{audit trail}) ایجاد می‌کند.

% ── 4.6 Evidence ──────────────────────────────────────────────────────────
\subsection{\lr{Evidence} — مدرک}

\textbf{دلیل وجود:}
مدرک‌ها انواع متفاوتی دارند (رونویسی، پزشکی/بیولوژیک، خودرو، مدرک هویتی).
طراحی با موجودیت پایه (\lr{Evidence}) و زیرموجودیت‌های تخصصی
(\lr{BioMedicalEvidence}، \lr{VehicleEvidence}، \lr{TranscriptionEvidence}، \lr{IdentityDocumentEvidence})
از الگوی \lr{Multi-table Inheritance} استفاده می‌کند که افزودن انواع جدید را بدون تغییر جدول پایه ممکن می‌سازد.

% ── 4.7 Trial & Punishment ────────────────────────────────────────────────
\subsection{\lr{Trial} و \lr{Punishment} — محاکمه و مجازات}

\textbf{دلیل وجود:}
پس از اتمام مرحله‌ی تحقیق، پرونده به دادگاه ارسال می‌شود.
\lr{Trial} رأی قاضی را ثبت می‌کند و \lr{Punishment} (رابطه‌ی یک‌به‌یک با محاکمه) جزئیات مجازات را نگه می‌دارد.
جداسازی این دو موجودیت به این دلیل است که امکان دارد محاکمه‌ای بدون صدور مجازات (حکم بی‌گناهی) به پایان برسد.

% ── 4.8 PaymentTransaction ────────────────────────────────────────────────
\subsection{\lr{PaymentTransaction} — تراکنش مالی}

\textbf{دلیل وجود:}
سه نوع پرداخت در سامانه وجود دارد: وثیقه (\lr{Bail})، جریمه (\lr{Fine}) و پاداش (\lr{Reward}).
یک موجودیت واحد با فیلد \lr{transaction\_type} به‌جای سه جدول جداگانه، گزارش‌گیری مالی یکپارچه را ساده‌تر می‌کند.

% ── 4.9 RewardReport ──────────────────────────────────────────────────────
\subsection{\lr{RewardReport} — گزارش پاداش}

\textbf{دلیل وجود:}
شهروندان می‌توانند اطلاعاتی درباره‌ی مظنونین ارائه دهند و پس از تأیید افسر و کارآگاه، پاداش دریافت کنند.
مقدار پاداش بر اساس امتیاز رتبه‌بندی مظنون (\lr{ranking\_score × 20,000,000}) محاسبه می‌شود.

% ── 4.10 CitizenTip ───────────────────────────────────────────────────────
\subsection{\lr{CitizenTip} — اطلاعیه‌ی شهروندی}

\textbf{دلیل وجود:}
جریان مستقل برای دریافت سرنخ از شهروندان در مورد مظنونین تحت تعقیب است.
کد رهگیری یکتا (\lr{unique\_tracking\_code}) پس از مفید شناخته شدن اطلاعیه صادر می‌شود تا شهروند بتواند وضعیت پاداش خود را پیگیری کند.

\section{پکیج‌های \lr{NPM} استفاده‌شده در پروژه}

\subsection{\lr{React} (\lr{v19})}
\textbf{کارکرد:} کتابخانه‌ی اصلی ساخت رابط کاربری مبتنی بر کامپوننت.\\
\textbf{توجیه:} اکوسیستم بزرگ، رندر اعلانی (\lr{declarative rendering})، پشتیبانی قوی از \lr{TypeScript}، و تناسب کامل با معماری تک‌صفحه‌ای (\lr{SPA}) که این پروژه نیاز دارد.

\subsection{\lr{React Router DOM} (\lr{v7})}
\textbf{کارکرد:} مسیریابی سمت کلاینت؛ تعریف مسیرها، پارامترهای پویا، و ریدایرکت‌ها.\\
\textbf{توجیه:} با داشتن بیش از ۱۰ نقش کاربری و صفحات متفاوت برای هر نقش، نیاز به یک سیستم مسیریابی اعلانی قوی اجتناب‌ناپذیر بود. نسخه ۷ قابلیت \lr{lazy loading} را نیز بهتر پشتیبانی می‌کند.

\subsection{\lr{Axios} (\lr{v1})}
\textbf{کارکرد:} کلاینت \lr{HTTP} برای ارتباط با \lr{REST API} بک‌اند.\\
\textbf{توجیه:} در مقایسه با \lr{fetch} بومی، \lr{interceptor}های \lr{Axios} مدیریت خودکار توکن \lr{JWT} و خطاهای \lr{401} را بسیار ساده‌تر می‌کنند.

\subsection{\lr{Zustand} (\lr{v5})}
\textbf{کارکرد:} مدیریت وضعیت سراسری سبک‌وزن.\\
\textbf{توجیه:} وضعیت احراز هویت و اطلاعات کاربر جاری باید در تمام کامپوننت‌ها در دسترس باشند. \lr{Zustand} بدون \lr{boilerplate} زیاد (در مقایسه با \lr{Redux}) این کار را انجام می‌دهد و با \lr{TypeScript} نیز به‌خوبی کار می‌کند.

\subsection{\lr{Vite} (\lr{v7})}
\textbf{کارکرد:} ابزار \lr{build} و سرور توسعه.\\
\textbf{توجیه:} سرعت بالای \lr{Hot Module Replacement}، زمان راه‌اندازی فوری، و پیکربندی ساده آن را به انتخاب پیش‌فرض اکثر پروژه‌های جدید \lr{React} تبدیل کرده است.

\subsection{\lr{Vitest} (\lr{v4})}
\textbf{کارکرد:} فریم‌ورک تست یکپارچه با \lr{Vite}.\\
\textbf{توجیه:} چون پروژه از \lr{Vite} استفاده می‌کند، \lr{Vitest} بدون تنظیمات اضافه در همان محیط \lr{build} اجرا می‌شود و سرعت اجرای تست‌ها را به‌طور قابل‌توجهی بالا می‌برد.

\section{نمونه کدهای تولیدشده توسط هوش مصنوعی}

\subsection{نمونه ۱ — سرویس احراز هویت در فرانت‌اند}

کد زیر توسط هوش مصنوعی تولید شده است و مدیریت توکن \lr{JWT} با \lr{Axios interceptor} را نشان می‌دهد:

\begin{latin}
\begin{lstlisting}[language=Java, caption={auth interceptor (TypeScript)}]
// src/services/api.ts
import axios from 'axios';
import { useAuthStore } from '../store/authStore';

const api = axios.create({ baseURL: '/api' });

api.interceptors.request.use((config) => {
  const token = useAuthStore.getState().accessToken;
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

api.interceptors.response.use(
  (res) => res,
  async (error) => {
    if (error.response?.status === 401) {
      await useAuthStore.getState().refresh();
      return api(error.config);
    }
    return Promise.reject(error);
  },
);

export default api;
\end{lstlisting}
\end{latin}

\subsection{نمونه ۲ — منطق سه ضربه در بک‌اند}

کد زیر توسط هوش مصنوعی پیشنهاد شد و منطق باطل‌شدن خودکار شکایت را پیاده‌سازی می‌کند:

\begin{latin}
\begin{lstlisting}[language=Python, caption={Complaint three-strike logic (Python)}]
# cases/models.py
def register_invalid_attempt(self):
    self.invalid_attempt_count += 1
    if self.invalid_attempt_count >= 3:
        self.status = self.Status.VOID
    else:
        self.status = self.Status.RETURNED
    self.save(update_fields=[
        "invalid_attempt_count", "status", "updated_at"
    ])
\end{lstlisting}
\end{latin}

\subsection{نمونه ۳ — کامپوننت محافظ مسیر بر اساس نقش}

\begin{latin}
\begin{lstlisting}[language=Java, caption={RoleGuard component (TypeScript/React)}]
// src/components/RoleGuard.tsx
import { Navigate } from 'react-router-dom';
import { useAuthStore } from '../store/authStore';

interface Props {
  allowedRoles: string[];
  children: React.ReactNode;
}

export function RoleGuard({ allowedRoles, children }: Props) {
  const { user } = useAuthStore();

  if (!user) return <Navigate to="/login" replace />;

  const hasRole = user.roles.some((r) => allowedRoles.includes(r));
  if (!hasRole) return <Navigate to="/403" replace />;

  return <>{children}</>;
}
\end{lstlisting}
\end{latin}

\begin{itemize}[rightmargin=1cm]

    ساختارهای رابط کاربری مانند \lr{RoleGuard}، \lr{lazy loading} مسیرها، و مدیریت وضعیت بارگذاری و خطا به‌خوبی پیشنهاد شدند.
  \item \textbf{رفع خطاهای تایپ \lr{TypeScript}:}
    تشخیص خطاهای نوع‌دهی و پیشنهاد نوع‌های صحیح برای \lr{props} و \lr{API response}ها دقیق بود.
  \item \textbf{نوشتن تست‌های واحد:}
    تولید تست‌های \lr{Vitest} برای کامپوننت‌های ساده سریع و کاربردی بود.
\end{itemize}

\subsection{ضعف‌ها}

\begin{itemize}[rightmargin=1cm]
  \item \textbf{درک ضعیف از منطق تجاری پیچیده:}
    صفحات مرتبط با جریان‌های چند مرحله‌ای (مانند تابلوی کارآگاه) نیاز به تصحیح‌های متعدد داشتند زیرا هوش مصنوعی زمینه‌ی کامل جریان کاری را نمی‌دانست.
  \item \textbf{ناسازگاری با ساختار \lr{API} واقعی:}
    کدهای تولیدشده اغلب فرض‌های اشتباهی درباره‌ی ساختار پاسخ \lr{API} داشتند که باید دستی اصلاح می‌شدند.
  \item \textbf{پیشنهاد کتابخانه‌های اضافی:}
    هوش مصنوعی گاهی پکیج‌های جدید (مانند \lr{react-query}) پیشنهاد می‌داد که با معماری انتخاب‌شده تعارض داشت.
\end{itemize}

\section{ضعف‌ها و قوت‌های هوش مصنوعی در توسعه‌ی بک‌اند}

\subsection{قوت‌ها}

\begin{itemize}[rightmargin=1cm]
  \item \textbf{طراحی مدل‌ها:}
    طراحی اولیه‌ی مدل‌های \lr{Django} (از جمله \lr{Multi-table Inheritance} برای مدرک‌ها) با کیفیت بالایی توسط هوش مصنوعی پیشنهاد شد.
  \item \textbf{نوشتن \lr{Serializer}ها و \lr{ViewSet}ها:}
    تولید \lr{serializer}های تودرتو و \lr{permission class}های سفارشی دقیق بود.
  \item \textbf{پیاده‌سازی \lr{State Machine}:}
    منطق ماشین وضعیت پرونده با جدول انتقال‌های صریح به‌درستی تولید شد.
  \item \textbf{نوشتن تست‌های یکپارچگی:}
    تست‌های \lr{Django REST Framework} با \lr{APIClient} برای جریان‌های کامل به‌خوبی تولید شدند.
\end{itemize}

\subsection{ضعف‌ها}

\begin{itemize}[rightmargin=1cm]
  \item \textbf{خطاهای مربوط به \lr{migration}:}
    پس از تغییرات مدل، هوش مصنوعی گاهی \lr{migration}های ناسازگار پیشنهاد می‌داد که باید دستی بررسی می‌شدند.
  \item \textbf{درک ناقص از \lr{RBAC} پویا:}
    منطق بررسی مجوزها با \lr{Group}-محور بودن سیستم گاهی اشتباه پیاده‌سازی می‌شد.
  \item \textbf{امنیت:}
    در برخی موارد هوش مصنوعی بررسی مالکیت شیء را فراموش می‌کرد (مثلاً بررسی اینکه آیا کاربر جاری مجاز به ویرایش یک مدرک خاص است).
\end{itemize}

\section{نیازسنجی‌های ابتدایی و نهایی پروژه}

\subsection{نیازسنجی ابتدایی}

در ابتدای پروژه، نیازهای اصلی به صورت زیر تعریف شدند:

\begin{enumerate}[rightmargin=1cm]
  \item سامانه‌ی ثبت و مدیریت شکایت شهروندان
  \item جریان تأیید چندمرحله‌ای (کادت $\to$ افسر)
  \item مدیریت پرونده‌ها با نقش‌های مختلف پلیسی
  \item زیرسامانه‌ی مدیریت مدارک و شواهد
  \item زیرسامانه‌ی مالی (وثیقه، جریمه، پاداش)
  \item واسط کاربری مبتنی بر نقش
\end{enumerate}

\subsection{نیازسنجی نهایی (تغییرات و اضافات)}

در طول توسعه، نیازهای زیر به پروژه اضافه یا اصلاح شدند:

\begin{enumerate}[rightmargin=1cm]
  \item \textbf{تأیید دوگانه برای پرونده‌های بحرانی:} پرونده‌های با سطح بحرانی (\lr{Critical}) نیاز به تأیید رئیس پلیس (\lr{Chief}) علاوه بر کاپیتان دارند.
  \item \textbf{سامانه‌ی مظنونین تحت تعقیب:} رتبه‌بندی مظنونین و کانال اطلاعیه‌ی شهروندی در نیازسنجی اولیه وجود نداشت.
  \item \textbf{تابلوی گرافی کارآگاه:} نیاز به ابزار بصری برای مدیریت ارتباط مدارک و مظنونین در طول توسعه مشخص شد.
  \item \textbf{زیرسامانه‌ی پزشکی قانونی:} گردش کار تأیید مدارک پزشکی توسط پزشک قانونی دیرتر اضافه شد.
\end{enumerate}

\subsection{قوت‌های تصمیمات گرفته‌شده}

\begin{itemize}[rightmargin=1cm]
  \item استفاده از \lr{RBAC} پویا با \lr{Group} جنگو: افزودن نقش‌های جدید بدون تغییر کد ممکن شد.
  \item طراحی \lr{State Machine} صریح برای پرونده: جلوگیری از انتقال‌های غیرمجاز به‌صورت خودکار.
  \item جداسازی \lr{apps} بر اساس \lr{domain}: انسجام بالای کد و آزمون‌پذیری بهتر.
\end{itemize}

\subsection{ضعف‌های تصمیمات گرفته‌شده}

\begin{itemize}[rightmargin=1cm]
  \item \textbf{یکپارچگی درگاه پرداخت:} در نیازسنجی ابتدایی یکپارچگی واقعی با درگاه در نظر گرفته نشد و همچنان ناقص است.
  \item \textbf{پیچیدگی مدل \lr{Case}:} متمرکز کردن تمام وضعیت‌ها در یک مدل واحد منجر به رشد ماشین وضعیت شد و نگهداری آن را سخت‌تر کرد.
\end{itemize}
